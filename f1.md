# Git Useful Commands (CLI):-

1. git init
2. git add f1.md
3. git restore --staged filename.extension 
4. git rm --cached f1.md

# Commit
1. git commit -m "commit message" 
2. git commit --amend --m"new commit Message"
3. git commit --amend --no-edit

# Rename a Branch
1. git branch -m master = respectfully
2. git branch -M master = forcefully
